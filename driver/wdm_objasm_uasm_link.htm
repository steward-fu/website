驅動程式 - Windows Driver Model (WDM) - 使用範例 - Assembly (ObjAsm) - 如何使用Linux UASM + MS Link.exe編譯驅動程式


$ uasm \
    -coff \
    -c \
    -q \
    -less \
    -nomlib \
    -FwNUL \
    -Zi \
    -Zd \
    main.asm

$ wine \
    "c:\\masm32\\bin\\link.exe" \
    main.o \
    /driver \
    /subsystem:native \
    /entry:DriverEntry \
    /base:0x10000 \
    "c:\\masm32\\lib\\wxp\\i386\\ntoskrnl.lib" \
    /out:main.sys
