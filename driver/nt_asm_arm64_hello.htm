驅動程式 - Windows NT Driver (Legacy) - 使用範例 - Assembly (ARM64) - Hello, world!


參考資訊：

main.asm 

    export DriverEntry
    extern DbgPrint

    area .data, data, arm64
Msg dcb "Hello, world!", 0

    area .text, code, arm64, align = 3
Unload
    ret

DriverEntry
    stp fp, lr, [sp, #-0x20]!
    mov fp, sp

    str x0, [sp, #0x10]
    str x1, [sp, #0x18]

    adrp x0, Msg
    bl DbgPrint

    ldr x1, [sp, #0x10]
    add x1, x1, #0x68
    adrp x0, Unload
    str x0, [x1]

    mov x0, #0
    ldp fp, lr, [sp], #0x20
    ret
    end
