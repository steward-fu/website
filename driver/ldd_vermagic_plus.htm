filename:       drivers/usb/serial/usbserial.ko

author:         Greg Kroah-Hartman <gregkh@linuxfoundation.org>
srcversion:     319AD7A689B7B86CC90F0E9
depends:        
intree:         Y
vermagic:       4.4.146+ SMP preempt mod_unload modversions aarch64
