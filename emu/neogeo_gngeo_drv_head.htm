typedef struct ROM_DEF {
    char name[32];
    char parent[32];
    char longname[128];
    uint32_t year;
    uint32_t romsize[10];
    uint32_t nb_romfile;

    struct romfile {
        char filename[32];
        uint8_t region;
        uint32_t src;
        uint32_t dest;
        uint32_t size;
        uint32_t crc;
    } rom[32];
} ROM_DEF;

P.S. parent指到另一個zip檔案，缺少的檔案會從parent zip提取，屬於階層式架構
