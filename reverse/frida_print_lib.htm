hello.c 

#include <stdio.h>
#include <unistd.h>

int main(int argc, char *argv[])
{
    while (1) {
        usleep(10000000);
    }

    return 0;
}

hook.py 

import frida

session = frida.attach("hello")

script = session.create_script("""
    rpc.exports.enumerateModules = () => {
      return Process.enumerateModules();
    };
""")

script.load()

for m in script.exports.enumerate_modules():
    print(m)
