main.py 

import os
import sys
from loguru import logger
from androguard.misc import AnalyzeAPK

logger.remove()
logger.add(lambda msg: None, level="INFO")

a, d, dx = AnalyzeAPK(sys.argv[1])

print("Activities:")
for act in a.get_activities():
    print(" -", act)

print("\nMain Activity:", a.get_main_activity())
