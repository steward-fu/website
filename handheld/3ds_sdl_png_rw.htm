main.c 

#include <stdio.h>
#include <stdlib.h>
#include <SDL.h>
#include <SDL_image.h>
 
#include "hex_png.h"
 
int main(void)
{
    SDL_Surface *screen = NULL;

    SDL_Init(SDL_INIT_VIDEO);
    screen = SDL_SetVideoMode(400, 240, 16, SDL_HWSURFACE);

    SDL_RWops *rw = SDL_RWFromMem(hex_png, sizeof(hex_png));
    SDL_Surface *png = IMG_Load_RW(rw, 1);
    SDL_BlitSurface(png, NULL, screen, NULL);
    SDL_Flip(screen);
    SDL_Delay(3000);

    SDL_FreeSurface(png);
    SDL_Quit();
    return 0;
}
