掌機 - Miyoo Mini Plus - C/C++ - SDL v2.0 - Save PNG


main.c 

#include <stdio.h>
#include <stdlib.h>
#include <SDL.h>
#include <SDL_image.h>
  
int main(int argc, char **argv)
{
    SDL_Surface *png = IMG_Load("main.png");
    IMG_SavePNG(png, "saved.png");
    SDL_FreeSurface(png);
    SDL_Quit();
    return 0;
}
