掌機 - TRIMUI BRICK - Build lrzsz


參考資訊：
https://repository.timesys.com/buildsources/l/lrzsz/lrzsz-0.12.20/

$ cd
$ wget https://github.com/steward-fu/website/releases/download/trimui-brick/brick-toolchain.tar.gz
$ tar xvf trimui-brick/brick-toolchain.tar.gz
$ sudo mv brick /opt/

$ cd
$ wget https://repository.timesys.com/buildsources/l/lrzsz/lrzsz-0.12.20/lrzsz-0.12.20.tar.gz
$ tar xvf lrzsz-0.12.20.tar.gz
$ cd lrzsz-0.12.20

$ export CROSS=/opt/brick/bin/aarch64-linux-gnu-
$ export CC=${CROSS}gcc
$ export AR=${CROSS}ar
$ export AS=${CROSS}as
$ export LD=${CROSS}ld
$ export CXX=${CROSS}g++
$ export HOST=arm-linux
$ 
$ ./configure
$ make -j4
