$ cd
$ wget https://github.com/steward-fu/website/releases/download/rs97/toolchain.7z

$ sudo mv rs97 /opt
$ export PATH=$PATH:/opt/rs97/bin

$ export TARGETMACH=mipsel-linux
$ export BUILDMACH=i686-pc-linux-gnu
$ export CROSS=mipsel-linux-
$ export CC=${CROSS}gcc
$ export LD=${CROSS}ld
$ export AS=${CROSS}as
$ export CXX=${CROSS}g++

$ wget https://github.com/steward-fu/website/releases/download/rs97/src_openmsx.7z
$ 7za x src_openmsx.7z
$ cd openmsx
$ rm -rf derived/3rdparty/src/libxml2-2.8.0
