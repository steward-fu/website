掌機 - Anbernic RG99 - Build ubiboot


$ cd
$ wget https://github.com/steward-fu/website/releases/download/rg99/bin_toolchain_rg99.tar.gz
$ tar xvf bin_toolchain_rg99.tar.gz
$ sudo mv rg99 /opt/
$ export PATH=$PATH:/opt/rg99/usr/bin

$ cd
$ wget https://github.com/steward-fu/website/releases/download/rg99/src_ubiboot.tar.gz
$ tar xvf src_ubiboot.tar.gz
$ cd ubiboot
$ make CONFIG=rg99 clean
$ make CONFIG=rg99 STAGE1_ONLY=y
$ mv output/rg99/ubiboot-v10.bin ubiboot-stage1.bin
