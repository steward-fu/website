int main(int argc, char** argv)
{
    struct js_event js;
    int fd = open("/dev/input/js1", O_RDONLY);
 
    while (1) {
        if (read(fd, &js, sizeof(struct js_event)) == sizeof(struct js_event)) {
            if (js.type == JS_EVENT_AXIS) {
                printf("%d: %d \n", js.number, js.value);
            }
        }
        usleep(30000);
    }
    return 0;
}
