$ cd
$ wget https://github.com/RT-Thread/toolchains-ci/releases/download/v1.7/riscv64-linux-musleabi_for_x86_64-pc-linux-gnu_latest.tar.bz2
$ tar xvf riscv64-linux-musleabi_for_x86_64-pc-linux-gnu_latest.tar.bz2
$ sudo mv riscv64-linux-musleabi_for_x86_64-pc-linux-gnu /opt
$ export PATH=/opt/riscv64-linux-musleabi_for_x86_64-pc-linux-gnu/bin:$PATH

$ cd
$ git clone https://github.com/RT-Thread/userapps.git
$ cd userapps
$ git clone -b rt-smart https://gitee.com/rtthread/rt-thread.git
$ source env.sh
$ cd rt-thread/bsp/allwinner/d1s

$ vim rt-thread/bsp/allwinner/libraries/drivers/drv_uart.c
    24   #define UART_DEFAULT_BAUDRATE 115200

$ vim rt-thread/bsp/allwinner/d1s/rtconfig.h
    44   #define RT_CONSOLE_DEVICE_NAME "uart4"
    264  #define UART4_TX_USING_GPIOB2
    265  #define UART4_RX_USING_GPIOB3

$ scons --menuconfig
$ scons -c
$ scons
$ sudo dd if=tools/boot0_sdcard_sun20iw1p1_d1s.bin of=/dev/sdX bs=1024 seek=8
$ sudo dd if=sd.bin of=/dev/sdX bs=1024 seek=56
