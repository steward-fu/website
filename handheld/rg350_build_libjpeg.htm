掌機 - Anbernic RG350 - Build libjpeg-turbo 2.0.2


$ cd
$ wget https://github.com/steward-fu/website/releases/download/rg350/bin_gcw0-toolchain.tar.gz
$ tar xvf bin_gcw0-toolchain.tar.gz
$ sudo mv gcw0-toolchain /opt
$ export PATH=$PATH:/opt/gcw0-toolchain/usr/bin

$ wget https://zh-tw.osdn.net/frs/g_redir.php?m=jaist&f=libjpeg-turbo%2F2.0.2%2Flibjpeg-turbo-2.0.2.tar.gz
$ tar xvf libjpeg-turbo-2.0.2.tar.gz
$ cd libjpeg-turbo-2.0.2
$ vim CMakeLists.txt +70
    set(CPU_TYPE mips)

$ mkdir build
$ cd build
$ CC=mipsel-linux-gcc CXX=mipsel-linux-g++ LD=mipsel-linux-ld cmake host=/opt/gcw0-toolchain/usr/shared/buildroot/cmake_toolchain_file.cmake ..
$ make -j4
$ make install
