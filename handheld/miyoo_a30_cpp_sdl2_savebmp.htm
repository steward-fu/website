掌機 - Miyoo A30 - C/C++ - SDL v2.0 - Save BMP


main.c 

#include <stdio.h>
#include <stdlib.h>
#include <SDL.h>
  
int main(int argc, char **argv)
{
    SDL_Surface *bmp = SDL_LoadBMP("main.bmp");
    SDL_SaveBMP(bmp, "saved.bmp");
    SDL_FreeSurface(bmp);
    SDL_Quit();
    return 0;
}

編譯、執行 

$ /opt/a30/bin/arm-linux-gcc \
    -I/opt/a30/arm-a30-linux-gnueabihf/sysroot/usr/include/SDL2 \
    -lSDL2 -lSDL2_gfx -lSDL2_image -lSDL2_ttf -lSDL2_mixer \
    -lGLESv2 main.c -o main
