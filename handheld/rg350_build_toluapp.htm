掌機 - Anbernic RG350 - Build toluapp


$ cd
$ wget https://github.com/steward-fu/website/releases/download/rg350/bin_gcw0-toolchain.tar.gz
$ tar xvf bin_gcw0-toolchain.tar.gz
$ sudo mv gcw0-toolchain /opt
$ export PATH=$PATH:/opt/gcw0-toolchain/usr/bin

$ export TARGETMACH=mipsel-linux
$ export BUILDMACH=i686-pc-linux-gnu
$ export CROSS=mipsel-linux
$ export CC=${CROSS}-gcc
$ export LD=${CROSS}-ld
$ export AS=${CROSS}-as
$ export CXX=${CROSS}-g++

$ wget https://github.com/steward-fu/website/releases/download/rg350/src_toluapp.tar.gz
$ tar xvf src_toluapp.tar.gz
$ cd toluapp
$ mkdir build
$ cd build
$ cmake ..
$ make
