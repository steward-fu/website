    .global _start

    .equiv PD,        (0x24 * 3)
    .equiv PE,        (0x24 * 4)

    .equiv LCD_RD,    (1 << 20)

delay:
    push {lr}
0:
    subs r0, #1
    bne 0b
    pop {pc}

lcd_rst:
    push {r4, r5, lr}
    ldr r4, =PIO_BASE + PE
    ldr r5, =0xffffffff

    bic r5, #LCD_RST
    str r5, [r4, #PIO_DATA]
    ldr r0, =10000
    bl delay

    orr r5, #LCD_RST
    str r5, [r4, #PIO_DATA]
    ldr r0, =10000
    bl delay
    pop {r4, r5, pc}

lcd_wr:
    push {r4, r5, lr}
    ldr r4, =PIO_BASE + PD

    and r2, r0, #0x00ff
    and r3, r0, #0xff00
    lsl r2, #1
    lsl r3, #2

    eor r5, r5
    orr r5, r1
    orr r5, r2
    orr r5, r3
    orr r5, #LCD_RD
    str r5, [r4, #PIO_DATA]

    orr r5, #LCD_WR
    str r5, [r4, #PIO_DATA]
    pop {r4, r5, pc}

lcd_dat:
    push {lr}
    mov r1, #LCD_RS
    bl lcd_wr
    pop {pc}

lcd_cmd:
    push {lr}
    mov r1, #0
    bl lcd_wr
    pop {pc}
    .end
