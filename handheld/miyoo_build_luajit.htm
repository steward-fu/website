掌機 - Miyoo - Build luajit 2.0.5


$ cd
$ wget https://github.com/steward-fu/website/releases/download/miyoo/env_toolchain.7z
$ 7za x env_toolchain.7z
$ sudo cp -a miyoo /opt/

$ wget http://luajit.org/download/LuaJIT-2.0.5.tar.gz
$ tar xvf LuaJIT-2.0.5.tar.gz
$ cd LuaJIT-2.0.5
$ sudo apt-get install g++-multilib libc6-dev-i386
$ make HOST_CC="gcc -m32" CROSS=/opt/miyoo/bin/arm-linux- TARGET_FLAGS="--sysroot /opt/miyoo/arm-miyoo-linux-uclibcgnueabi/sysroot -ldl" TARGET_SYS=arm
