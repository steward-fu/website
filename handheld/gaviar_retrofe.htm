編譯 

$ cmake \
    RetroFE/Source \
    -BRetroFE/Build \
    -DVERSION_MAJOR=0 \
    -DVERSION_MINOR=0 \
    -DVERSION_BUILD=0 \
    -DCMAKE_TOOLCHAIN_FILE=/opt/gaviar/share/buildroot/toolchainfile.cmake

$ cmake --build RetroFE/Build
$ file RetroFE/Build/retrofe 
    RetroFE/Build/retrofe: ELF 64-bit LSB pie executable, UCB RISC-V

複製到MicroSD 

$ mkdir /xxx/SDCARD/RetroFE
$ cp RetroFE/Build/retrofe /xxx/SDCARD/RetroFE
$ cp -a ~/FunKey-OS/FunKey/board/funkey/rootfs-overlay/usr/games/* /xxx/SDCARD/RetroFE

修改設定 

$ vim /xxx/SDCARD/RetroFE/settings.conf
    horizontal = 320
    baseItemPath=/boot/roms

開機後執行 

$ mount -o remount,rw /boot
$ cd /boot/RetroFE
$ SDL_NOMOUSE=1 SDL_VIDEODRIVER=fbcon ./retrofe

完成
