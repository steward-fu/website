main.c 

#include <stdio.h>
#include <stdlib.h>
#include <SDL.h>
 
int main(void)
{
    SDL_Surface *bmp = SDL_LoadBMP("sdmc:/3ds/main.bmp");
    SDL_SaveBMP(bmp, "sdmc:/3ds/saved.bmp");
    SDL_FreeSurface(bmp);
    SDL_Quit();
    return 0;
}
