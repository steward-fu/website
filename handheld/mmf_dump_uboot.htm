掌機 - Miyoo Mini Flip - Dump U-Boot Env


dump.py 

import os
import sys

FILE = sys.argv[1]

START = 0x5f004
END   = 0x60000

with open(FILE, "rb") as f:
    f.seek(START)
    data = f.read(END - START)

offset = START

for item in data.split(b"\0"):
    if not item:
        offset += 1
        continue

    try:
        s = item.decode("utf-8")
    except UnicodeDecodeError:
        s = item.decode("latin-1")

    print(f"{s}")
    offset += len(item) + 1
