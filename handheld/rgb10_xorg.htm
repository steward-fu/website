Section "Device"
    Identifier "Framebuffer"
    Driver "fbdev"
    Option "fbdev" "/dev/fb0"
EndSection
 
Section "Monitor"
    Identifier "Generic Monitor"
    ModelName "Monitor Model"
    DisplaySize 320 480
EndSection

Section "Screen"
    Identifier "Default Screen"
    Device "Framebuffer"
    Monitor "Generic Monitor"
EndSection
