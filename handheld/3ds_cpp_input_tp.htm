main.c 

#include <3ds.h>
#include <stdio.h>
 
int main(void)
{
    touchPosition touch;
 
    gfxInitDefault();
    consoleInit(GFX_TOP, NULL);
    while (aptMainLoop()) {
        hidScanInput();
        hidTouchRead(&touch);
        
        consoleClear();
        printf("touch, x:%d, y:%d\n", touch.px, touch.py);
 
        gfxFlushBuffers();
        gfxSwapBuffers();
        gspWaitForVBlank();
    }
    gfxExit();
    return 0;
}
