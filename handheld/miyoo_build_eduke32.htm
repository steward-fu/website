$ sudo cp -a miyoo /opt/

$ export PATH=$PATH:/opt/miyoo/bin
$ export TARGETMACH=arm-linux
$ export BUILDMACH=i686-pc-linux-gnu
$ export CROSS=arm-linux
$ export CC=${CROSS}-gcc
$ export LD=${CROSS}-ld
$ export AS=${CROSS}-as
$ export CXX=${CROSS}-g++

$ wget https://github.com/steward-fu/website/releases/download/miyoo/src_eduke32.7z

$ cd eduke32

  WITHOUT_GTK=1
