系統 - Debian - Memory - 列出耗費記憶體的Process


mem.sh 

#!/bin/bash
for p in /proc/[0-9]*; do
    pid=${p##*/}
    rss=$(awk '/^VmRSS:/ {print $2}' "$p/status" 2>/dev/null)
    [ -n "$rss" ] && printf "%10s %10s KB  " "$pid" "$rss" && cat "$p/comm" 2>/dev/null
done | sort -nrk 2 | head -20
