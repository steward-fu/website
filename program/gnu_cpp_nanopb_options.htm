main.proto 

syntax = "proto3";

message Test {
    string s1 = 1;
    int32  v1 = 2;
    string s2 = 3;
    int32  v2 = 4;
}

main.options 

Test.s1 max_size:255
Test.s2 max_size:255

main.c 

#include <stdio.h>
#include <unistd.h>
#include <fcntl.h>
#include <pb_encode.h>
#include <pb_decode.h>
 
#include "main.pb.h"
 
int main(int argc, char **argv)
{
    char buf[255] = {0};
     
    Test org = Test_init_zero;
    pb_ostream_t ostream = pb_ostream_from_buffer(buf, sizeof(buf));
    strcpy(org.s1, "1");
    org.v1 = 100;
    strcpy(org.s2, "2");
    org.v2 = 200;
    pb_encode(&ostream, Test_fields, &org);

    Test out = Test_init_zero;
    pb_istream_t istream = pb_istream_from_buffer(buf, sizeof(buf));
    pb_decode(&istream, Test_fields, &out);
 
    printf("s1 %s\n", out.s1);
    printf("v1 %d\n", out.v1);
    printf("s2 %s\n", out.s2);
    printf("v2 %d\n", out.v2);
    return 0;
}

編譯、執行 

$ protoc --nanopb_out=. main.proto
$ gcc main.c -o test main.pb.c -I. -lprotobuf-nanopb
$ ./test
