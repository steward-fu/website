main.cpp 

#include <math.h>
#include <FL/Fl.H>
#include <FL/Fl_Gl_Window.H>
#include <FL/Fl_Window.H>
#include <FL/gl.h>

class MyGL : public Fl_Gl_Window {
public:
    MyGL(int x, int y, int w, int h) : Fl_Gl_Window(x, y, w, h) {
    }

    void draw() override {
        if (!valid()) {
            GLdouble nearPlane = 0.1;
            GLdouble farPlane = 100.0;
            GLdouble top = nearPlane * tan(45.0 * 3.14159 / 360.0);
            GLdouble right = top * ((GLfloat)w() / (GLfloat)h());

            glViewport(0, 0, w(), h());
            glMatrixMode(GL_PROJECTION);
            glFrustum(-right, right, -top, top, nearPlane, farPlane);
           
            glMatrixMode(GL_MODELVIEW);
            glLoadIdentity();
        }

        glLoadIdentity();
        glTranslatef(-1.0f, 0.0f, -5.0f);
        glBegin(GL_TRIANGLES);
            glColor3f(1.0f, 0.0f, 0.0f);
            glVertex3f(0.0f, 0.5f, 0.0f);
       
            glColor3f(0.0f, 1.0f, 0.0f);
            glVertex3f(-0.5f, -0.5f, 0.0f);
       
            glColor3f(0.0f, 0.0f, 1.0f);
            glVertex3f(0.5f, -0.5f, 0.0f);
        glEnd();
       
        glLoadIdentity();
        glTranslatef(1.0f, 0.0f, -5.0f);
        glBegin(GL_QUADS);
            glColor3f(1.0f, 0.0f, 0.0f);
            glVertex3f(1.0f, 1.0f, 0.0f);
       
            glColor3f(0.0f, 1.0f, 0.0f);
            glVertex3f(-1.0f, 1.0f, 0.0f);
       
            glColor3f(0.0f, 0.0f, 1.0f);
            glVertex3f(-1.0f, -1.0f, 0.0f);
       
            glColor3f(1.0f, 1.0f, 0.0f);
            glVertex3f(1.0f, -1.0f, 0.0f);
        glEnd();
    }
};

int main() {
    const int w = 320;
    const int h = 240;

    Fl_Window window(w, h, "main");
    MyGL mygl(0, 0, w, h);

    window.end();
    window.show();

    return Fl::run();
}
