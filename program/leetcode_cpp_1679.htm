class Solution {
public:
    int maxOperations(vector<int>& nums, int k) {
        int r = 0;
        int st = 0;
        int ed = nums.size() - 1;
        std::sort(nums.begin(), nums.end());

        while (st < ed) {
            int v = nums[st] + nums[ed];

            if (v == k) {
                st += 1;
                ed -= 1;
                r += 1;
            }
            else if (v < k) {
                st += 1;
            }
            else if (v > k) {
                ed -= 1;
            }
        }

        return r;
    }
};
