Windows的視窗可以分成：Dialog、Single Document Interface(SDI)、Multiple Document Interface(MDI)三種類別，相較於SDI、MDI這兩類別的程式，Dialog類別是比較容易上手學習的一個類別，相當適合初學者入門學習，而Windows程式設計的精髓就是，學會其中一個類別，其餘類別都可以很容易上手，Windows控制項(如：Button、Edit)也是此類道理，比較需要注意的是，Windows視窗程式設計的核心是圍繞着事件處理(處理Message)，任何的視窗設定或者溝通都是以事件導向為原則，因此，寫出一個簡單的視窗框架，將有助於了解Windows視窗程式的運作

main.S 

    .global WinMain

    .equiv WC_DIALOG, 0x8002
    .equiv WM_DESTROY, 2
    .equiv WM_CLOSE, 0x10
    .equiv WS_OVERLAPPED, 0
    .equiv WS_CAPTION, 0xc00000
    .equiv WS_SYSMENU, 0x80000
    .equiv WS_THICKFRAME, 0x40000
    .equiv WS_MINIMIZEBOX, 0x20000
    .equiv WS_MAXIMIZEBOX, 0x10000
    .equiv WS_VISIBLE, 0x10000000
    .equiv WS_EX_LEFT, 0x00000000
    .equiv WS_OVERLAPPEDWINDOW, (WS_OVERLAPPED | WS_CAPTION | WS_SYSMENU | WS_THICKFRAME | WS_MINIMIZEBOX | WS_MAXIMIZEBOX)

    .data
szMsg: .skip 28
szAppName: .asciz "main" 

    .text
WinMain:
    push { lr }
 
    ldr r3, =0
    ldr r2, =0
    ldr r1, =0
    ldr r0, =0
    push { r0, r1, r2, r3 }
    ldr r3, =320
    ldr r2, =240
    ldr r1, =0
    ldr r0, =0
    push { r0, r1, r2, r3 }
    ldr r3, =WS_OVERLAPPEDWINDOW | WS_VISIBLE
    ldr r2, =szAppName
    ldr r1, =WC_DIALOG
    ldr r0, =WS_EX_LEFT
    bl CreateWindowExA
    add sp, sp, #32
 
loop:
    ldr r3, =0
    ldr r2, =0
    ldr r1, =0
    ldr r0, =szMsg
    bl GetMessageA
 
    cmp r0, #0
    beq exit
   
    ldr r0, =szMsg
    bl DispatchMessageA
    b loop
 
exit:
    pop { pc }

編譯、執行 

$ winegcc-armhf -b arm-linux-gnueabihf main.S -o main.exe
$ WINEPREFIX=~/.wine_arm32 myarm32 main.exe.so
