class Solution {
public:
    double findMaxAverage(vector<int>& nums, int k) {
        int r = -INT_MAX;
        int sum = 0;

        for (int i = 0; i < nums.size(); i++) {
            sum += nums[i];

            if (i >= (k - 1)) {
                if (sum > r) {
                    r = sum;
                }
                sum -= nums[i - (k - 1)];
            }
        }

        return (double)r / k;
    }
};
