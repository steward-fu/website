bool* kidsWithCandies(int* candies, int candiesSize, int extraCandies, int* returnSize)
{
    int cc = 0;
    int max = 0;
    bool *r = malloc(sizeof(bool) * candiesSize);

    for (cc = 0; cc < candiesSize; cc++) {
        if (candies[cc] > max) {
            max = candies[cc];
        }
        r[cc] = false;
    }

    *returnSize = candiesSize;
    for (cc = 0; cc < candiesSize; cc++) {
        if ((candies[cc] + extraCandies) >= max) {
            r[cc] = true;
        }
    }

    return r;
}
