int pivotIndex(int* nums, int numsSize)
{
    int cc = 0;
    int sum = 0;
    int left = 0;
    int check = 0;

    for (cc = 0; cc < numsSize; cc++) {
        sum += nums[cc];
    }

    for (cc = 0; cc < numsSize; cc++) {
        check = sum - nums[cc] - left;

        if (left == check) {
            return cc;
        }

        left += nums[cc];
    }

    return -1;
}
