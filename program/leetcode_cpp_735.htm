class Solution {
public:
    vector<int> asteroidCollision(vector<int>& asteroids) {
        vector<int> r;

        for (auto it = asteroids.begin(); it != asteroids.end(); it++) {
            if (*it > 0) {
                r.push_back(*it);
                continue;
            }

            if ((r.size() == 0) || (*r.rbegin() < 0)) {
                r.push_back(*it);
                continue;
            }

            if (-*it >= *r.rbegin()) {
                if (-*it > *r.rbegin()) {
                    it--;
                }
                r.pop_back();
            }
        }

        return r;
    }
};
