class Solution {
public:
    bool uniqueOccurrences(vector<int>& arr) {
        unordered_set<int> m1;
        unordered_map<int, int> m0;

        for (int n : arr) {
            m0[n] += 1;
        }

        for (auto n : m0) {
            if (m1.count(n.second) > 0) {
                return false;
            }
            m1.insert(n.second);
        }

        return true;
    }
};
