int singleNumber(int* nums, int numsSize)
{
    int cc = 0;
    int cnt[60000] = { 0 };

    for (cc = 0; cc < numsSize; cc++) {
        cnt[nums[cc] + 30000] += 1;
    }

    for (cc = 0; cc < 60000; cc++) {
        if (cnt[cc] == 1) {
            return cc - 30000;
        }
    }

    return 0;
}
