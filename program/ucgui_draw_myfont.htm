int main(int argc, char *argv[])
{
    GUI_Init();
    GUI_SetColor(GUI_WHITE);

    GUI_UC_SetEncodeUTF8();
    GUI_SetFont(&myFont);
    GUI_DispString("GUI字體測試");

    GUI_Delay(3000);
    return 0;
}

$ ./main
