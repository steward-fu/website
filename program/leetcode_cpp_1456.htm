class Solution {
public:
    int maxVowels(string s, int k) {
        int r = 0;
        int cnt = 0;
        vector<char> vows = { 'a', 'e', 'i', 'o', 'u' };

        for (int i = 0; i < k; i++) {
            if (std::find(vows.begin(), vows.end(), s[i]) != vows.end()) {
                cnt += 1;
            }
        }

        r = cnt;
        for (int i = k; i < s.size(); i++) {
            if (std::find(vows.begin(), vows.end(), s[i]) != vows.end()) {
                cnt += 1;
            }

            if (std::find(vows.begin(), vows.end(), s[i - k]) != vows.end()) {
                cnt -= 1;
            }

            if (cnt > r) {
                r = cnt;
            }
        }

        return r;
    }
};
