class Solution {
public:
    string predictPartyVictory(string senate) {
        queue<int> r;
        queue<int> d;
        int size = senate.size();

        for (int i = 0; i < size; i++) {
            if (senate[i] == 'R') {
                r.push(i);
            }
            else {
                d.push(i);
            }
        }

        while (!r.empty() && !d.empty()) {
            int i = r.front();
            int j = d.front();

            r.pop();
            d.pop();
            if (i < j) {
                r.push(i + size);
            }
            else {
                d.push(j + size);
            }
        }

        return r.size() > d.size() ? "Radiant" : "Dire";
    }
};
