main.c 

#include "GUI.h"
#include "EDIT.h"
 
int main(int argc, char *argv[])
{
    EDIT_Handle hEdit = 0;
 
    GUI_Init();
    GUI_SetBkColor(GUI_GRAY);
    GUI_Clear();

    hEdit = EDIT_Create(100, 100, 100, 18, 100, 128, 0);
    EDIT_SetText(hEdit, "Test");
    EDIT_SetFont(hEdit, &GUI_Font8x16);
    EDIT_SetTextColor(hEdit, 0, GUI_BLACK);
    WM_SetFocus(hEdit);
    GUI_Delay(1000);
    EDIT_Delete(hEdit);
 
    GUI_Delay(3000);
    return 0;
}

編譯、執行 

$ gcc main.c -o main libucgui.a -IGUI_X -IGUI/Core -IGUI/Widget -IGUI/WM -lSDL
$ ./main
