main.asm 

    .386
    .model flat,stdcall
    option casemap : none
  
include c:\masm32\include\windows.inc
include c:\masm32\include\ogl\kernel32.inc
include c:\masm32\include\ogl\user32.inc
include c:\masm32\include\ogl\gdi32.inc
include c:\masm32\include\ogl\gl.def
include c:\masm32\include\ogl\glu.def
include c:\masm32\include\ogl\winextra\winextra.def
  
includelib c:\masm32\lib\gdi32.lib
includelib c:\masm32\lib\glu32.lib
includelib c:\masm32\lib\user32.lib
includelib c:\masm32\lib\opengl32.lib
includelib c:\masm32\lib\kernel32.lib

TEXCOORD MACRO u, v
    push fpc(v)
    push fpc(u)
    push GL_TEXTURE0_ARB
    call dword ptr [pglMultiTexCoord2fARB]

    push fpc(v)
    push fpc(u)
    push GL_TEXTURE1_ARB
    call dword ptr [pglMultiTexCoord2fARB]
ENDM

    .data
szCaption               db "main", 0
hWin                    dd 0
hInstance               dd 0
CommandLine             dd 0
defWndProc              dd 0
hDC                     HDC 0
hRC                     HGLRC 0
nearPlane               real8 0.1
farPlane                real8 100.0
angle                   real8 45.0
szBMP0Name              db "main.bmp", 0
szBMP1Name              db "bump.bmp", 0
szBMPNames              dd offset szBMP0Name, offset szBMP1Name
tex                     GLuint 2 dup(0)
pglMultiTexCoord2fARB   dd 0
pglActiveTextureARB     dd 0
szMultiTexCoord2f       db "glMultiTexCoord2fARB", 0
szActiveTexture         db "glActiveTextureARB", 0

    .code
WndProc proc hWnd : HWND, uMsg : UINT, wParam : WPARAM, lParam : LPARAM
    .if uMsg == WM_CLOSE
        invoke DestroyWindow, hWnd
        xor eax, eax
        ret
    .elseif uMsg == WM_DESTROY
        invoke PostQuitMessage, 0
        xor eax, eax
        ret
    .endif
   
    invoke CallWindowProc, defWndProc, hWnd, uMsg, wParam, lParam
    ret
WndProc endp
    
WinMain proc hInst : DWORD, hPrevInst : DWORD, CmdLine : DWORD, CmdShow : DWORD
    local i : DWORD
    local w : DWORD
    local h : DWORD
    local l : DWORD
    local msg : MSG
    local fmt : DWORD
    local pfd : PIXELFORMATDESCRIPTOR
    local top : real8
    local right : real8
    local mtop : real8
    local mright : real8
    local bm : BITMAP
    local bmi : BITMAPINFO
    local bmp : HBITMAP
    local pixels : DWORD
 
    mov w, 320
    mov h, 240
    invoke CreateWindowEx, WS_EX_LEFT, WC_DIALOG, offset szCaption, WS_OVERLAPPEDWINDOW or WS_VISIBLE, 0, 0, w, h, NULL, NULL, NULL, NULL
    mov hWin, eax
    
    invoke SetWindowLong, hWin, GWL_WNDPROC, WndProc
    mov defWndProc, eax
  
    invoke RtlZeroMemory, addr pfd, sizeof PIXELFORMATDESCRIPTOR
    mov pfd.nSize, sizeof PIXELFORMATDESCRIPTOR
    mov pfd.nVersion, 1
    mov pfd.dwFlags, PFD_DRAW_TO_WINDOW or PFD_SUPPORT_OPENGL or PFD_DOUBLEBUFFER
    mov pfd.iPixelType, PFD_TYPE_RGBA
    mov pfd.cColorBits, 16
    mov pfd.cDepthBits, 16
    mov pfd.dwLayerMask, PFD_MAIN_PLANE
  
    invoke GetDC, hWin
    mov hDC, eax
    invoke ChoosePixelFormat, hDC, addr pfd
    mov fmt, eax
    invoke SetPixelFormat, hDC, fmt, addr pfd
    invoke wglCreateContext, hDC
    mov hRC, eax
    invoke wglMakeCurrent, hDC, hRC

    invoke wglGetProcAddress, offset szMultiTexCoord2f
    mov pglMultiTexCoord2fARB, eax

    invoke wglGetProcAddress, offset szActiveTexture
    mov pglActiveTextureARB, eax

    mov i, 0
    invoke glGenTextures, 2, offset tex
@@load_loop:
    mov eax, i
    invoke LoadImageA, NULL, szBMPNames[eax * 4], IMAGE_BITMAP, 0, 0, LR_LOADFROMFILE
    mov bmp, eax
    invoke GetObject, bmp, sizeof BITMAP, addr bm

    invoke RtlZeroMemory, addr bmi, sizeof BITMAPINFO 
    mov bmi.bmiHeader.biSize, sizeof BITMAPINFOHEADER
    push bm.bmWidth
    pop bmi.bmiHeader.biWidth
    mov eax, -1
    mul bm.bmHeight
    mov bmi.bmiHeader.biHeight, eax
    mov bmi.bmiHeader.biPlanes, 1
    mov bmi.bmiHeader.biBitCount, 24
    mov bmi.bmiHeader.biCompression, BI_RGB
 
    mov eax, 3
    mul bm.bmWidth
    mul bm.bmHeight 
    mov l, eax
    invoke GetProcessHeap
    invoke HeapAlloc, eax, 0, l
    mov pixels, eax
    invoke GetDIBits, hDC, bmp, 0, bm.bmHeight, pixels, addr bmi, DIB_RGB_COLORS

    .if i == 0
        push GL_TEXTURE0_ARB
    .else
        push GL_TEXTURE1_ARB
    .endif
    call DWORD PTR [pglActiveTextureARB]

    invoke glEnable, GL_TEXTURE_2D
    mov eax, i
    invoke glBindTexture, GL_TEXTURE_2D, tex[eax * 4]
    invoke glTexImage2D, GL_TEXTURE_2D, 0, GL_RGB, bm.bmWidth, bm.bmHeight, 0, GL_BGR, GL_UNSIGNED_BYTE, pixels
    invoke glTexParameteri, GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR
    invoke glTexParameteri, GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR
    invoke glTexEnvi, GL_TEXTURE_ENV, GL_TEXTURE_ENV_MODE, GL_MODULATE
 
    invoke GetProcessHeap
    invoke HeapFree, eax, 0, pixels
    invoke DeleteObject, bmp
 
    inc i
    cmp i, 2
    jb @@load_loop

    fld angle
    fldpi
    fmulp st(1), st(0)
  
    mov eax, 360
    push eax
    fild dword ptr [esp]
    add esp, 4
  
    fdivp st(1), st(0)
  
    fptan
    fstp st(0)
  
    fmul nearPlane
    fstp top
  
    fld top
  
    push w
    fild dword ptr [esp]
    add esp, 4
  
    push h
    fild dword ptr [esp]
    add esp, 4
  
    fdivp st(1), st(0)
    fmulp st(1), st(0)
    fstp right
  
    fld top
    fchs
    fstp mtop
  
    fld right
    fchs
    fstp mright
  
    invoke glViewport, 0, 0, w, h
    invoke glMatrixMode, GL_PROJECTION
    invoke glFrustum, \
        dword ptr [mright + 0], \
        dword ptr [mright + 4], \
        dword ptr [right + 0], \
        dword ptr [right + 4], \
        dword ptr [mtop + 0], \
        dword ptr [mtop + 4], \
        dword ptr [top + 0], \
        dword ptr [top + 4], \
        dword ptr [nearPlane + 0], \
        dword ptr [nearPlane + 4], \
        dword ptr [farPlane + 0], \
        dword ptr [farPlane + 4]
  
    invoke glMatrixMode, GL_MODELVIEW
    invoke glLoadIdentity
  
@@:
    invoke GetMessage, addr msg, NULL, 0, 0
    cmp eax, 0
    je @f
    invoke DispatchMessage, addr msg

    invoke glLoadIdentity
    invoke glTranslatef, fpc(0.0), fpc(0.0), fpc(-5.0)
    invoke glRotatef, fpc(45.0), fpc(0.0), fpc(1.0), fpc(0.0)

    invoke glBegin, GL_QUADS
        TEXCOORD 0.0, 1.0
        invoke glVertex3f, fpc(-1.0), fpc(-1.0), fpc(1.0)
        TEXCOORD 1.0, 1.0
        invoke glVertex3f, fpc(1.0), fpc(-1.0), fpc(1.0)
        TEXCOORD 1.0, 0.0
        invoke glVertex3f, fpc(1.0), fpc(1.0), fpc(1.0)
        TEXCOORD 0.0, 0.0
        invoke glVertex3f, fpc(-1.0), fpc(1.0), fpc(1.0)

        TEXCOORD 0.0, 0.0
        invoke glVertex3f, fpc(-1.0), fpc(-1.0), fpc(-1.0)
        TEXCOORD 0.0, 1.0
        invoke glVertex3f, fpc(-1.0), fpc(1.0), fpc(-1.0)
        TEXCOORD 1.0, 1.0
        invoke glVertex3f, fpc(1.0), fpc(1.0), fpc(-1.0)
        TEXCOORD 1.0, 0.0
        invoke glVertex3f, fpc(1.0), fpc(-1.0), fpc(-1.0)

        TEXCOORD 1.0, 1.0
        invoke glVertex3f, fpc(-1.0), fpc(1.0), fpc(-1.0)
        TEXCOORD 1.0, 0.0
        invoke glVertex3f, fpc(-1.0), fpc(1.0), fpc(1.0)
        TEXCOORD 0.0, 0.0
        invoke glVertex3f, fpc(1.0), fpc(1.0), fpc(1.0)
        TEXCOORD 0.0, 1.0
        invoke glVertex3f, fpc(1.0), fpc(1.0), fpc(-1.0)

        TEXCOORD 0.0, 1.0
        invoke glVertex3f, fpc(-1.0), fpc(-1.0), fpc(-1.0)
        TEXCOORD 1.0, 1.0
        invoke glVertex3f, fpc(1.0), fpc(-1.0), fpc(-1.0)
        TEXCOORD 1.0, 0.0
        invoke glVertex3f, fpc(1.0), fpc(-1.0), fpc(1.0)
        TEXCOORD 0.0, 0.0
        invoke glVertex3f, fpc(-1.0), fpc(-1.0), fpc(1.0)

        TEXCOORD 0.0, 0.0
        invoke glVertex3f, fpc(1.0), fpc(-1.0), fpc(-1.0)
        TEXCOORD 0.0, 1.0
        invoke glVertex3f, fpc(1.0), fpc(1.0), fpc(-1.0)
        TEXCOORD 1.0, 1.0
        invoke glVertex3f, fpc(1.0), fpc(1.0), fpc(1.0)
        TEXCOORD 1.0, 0.0
        invoke glVertex3f, fpc(1.0), fpc(-1.0), fpc(1.0)

        TEXCOORD 1.0, 0.0
        invoke glVertex3f, fpc(-1.0), fpc(-1.0), fpc(-1.0)
        TEXCOORD 0.0, 0.0
        invoke glVertex3f, fpc(-1.0), fpc(-1.0), fpc(1.0)
        TEXCOORD 0.0, 1.0
        invoke glVertex3f, fpc(-1.0), fpc(1.0), fpc(1.0)
        TEXCOORD 1.0, 1.0
        invoke glVertex3f, fpc(-1.0), fpc(1.0), fpc(-1.0)
    invoke glEnd

    invoke SwapBuffers, hDC
  
    jmp @b
@@:
    invoke wglMakeCurrent, NULL, NULL
    invoke wglDeleteContext, hRC
    invoke ReleaseDC, hWin, hDC
    invoke glDeleteTextures, 2, offset tex
 
    mov eax, msg.wParam
    ret
WinMain endp
   
start:
    invoke GetModuleHandle, NULL
    mov hInstance, eax
     
    invoke GetCommandLine
    mov CommandLine, eax
     
    invoke WinMain, hInstance, NULL, CommandLine, SW_SHOWDEFAULT
    xor eax, eax
    ret
end start
