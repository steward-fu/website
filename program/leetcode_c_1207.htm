bool uniqueOccurrences(int* arr, int arrSize)
{
    #define MAX_SIZE 2001

    int i = 0;
    int m0[MAX_SIZE] = { 0 };
    int m1[MAX_SIZE] = { 0 };

    for (i = 0; i < arrSize; i++) {
        m0[arr[i] + 1000] += 1;
    }

    for (i = 0; i < MAX_SIZE; i++) {
        if (m0[i] && m1[m0[i]]) {
            return false;
        }
        m1[m0[i]] = 1;
    }

    return true;
}
