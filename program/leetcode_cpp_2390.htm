class Solution {
public:
    string removeStars(string s) {
        std::deque<char> r;

        for (char ch : s) {
            if (ch == '*') {
                r.pop_back();
            }
            else {
                r.push_back(ch);
            }
        }

        return std::string(r.begin(), r.end());
    }
};
