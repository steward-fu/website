int minOperations(int* nums, int numsSize)
{
    int r = 0;
    int cc = 0;

    for (cc = 0; cc < numsSize; cc+= 1) {
        if (nums[cc] == 0) {
            if ((cc + 2) >= numsSize) {
                return -1;
            }
            nums[cc + 1] ^= 1;
            nums[cc + 2] ^= 1;
            r += 1;
        }
    }
    return r;
}
