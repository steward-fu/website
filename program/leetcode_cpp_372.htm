class Solution {
public:
    int superPow(int a, vector<int>& b) {
        using ll = long long;
        const int MOD = 1337;

        auto pow = [](ll base, int e) {
            ll ans = 1;

            while (e > 0) {
                if (e & 1) {
                    ans = (ans * base) % MOD;
                }
                base = base * base % MOD;
                e >>= 1;
            }
    
            return ans;
        };

        ll ans = 1;
        for (int i = b.size() -1; i >= 0; --i) {
            ans = (ans * pow(a, b[i])) % MOD;
            a = pow(a, 10);
        }

        return ans;
    }
};
