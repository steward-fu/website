int idx = 0;
char *p = NULL;

void push(char ch)
{
    p[idx++] = ch;
}

void pop(void)
{
    if (idx > 0) {
        idx -= 1;
        p[idx] = 0;
    }
}

char* removeStars(char *s)
{
    int c0 = 0;
    int len = strlen(s);

    p = malloc(len + 1);
    memset(p, 0, len + 1);

    for (c0 = 0; c0 < len; c0++) {
        if (s[c0] != '*') {
            push(s[c0]);
        }
        else {
            pop();
        }
    }
    return p;
}

void push(char *p, int *idx, char ch)
{
    p[(*idx)++] = ch;
}

void pop(char *p, int *idx)
{
    if ((*idx) > 0) {
        (*idx) -= 1;
        p[*idx] = 0;
    }
}

char* removeStars(char *s)
{
    int c0 = 0;
    int idx = 0;
    int len = strlen(s);
    char *p = malloc(len + 1);

    memset(p, 0, len + 1);
    for (c0 = 0; c0 < len; c0++) {
        if (s[c0] != '*') {
            push(p, &idx, s[c0]);
        }
        else {
            pop(p, &idx);
        }
    }
    return p;
}
