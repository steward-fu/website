解答： 

class Solution {
public:
    vector<int> pivotArray(vector<int>& nums, int pivot) {
        vector<int> less;
        vector<int> equal;
        vector<int> large;

        for (int n : nums) {
            if (n == pivot) {
                equal.push_back(n);
            }
            else if (n < pivot) {
                less.push_back(n);
            }
            else {
                large.push_back(n);
            }
        }

        vector<int> ans;

        ans.insert(ans.begin(), less.begin(), less.end());
        ans.insert(ans.end(), equal.begin(), equal.end());
        ans.insert(ans.end(), large.begin(), large.end());

        return ans;
    }
};
