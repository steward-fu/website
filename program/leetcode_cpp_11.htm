class Solution {
public:
    int maxArea(vector<int>& height) {
        int r = 0;
        int st = 0;
        int ed = height.size() - 1;

        while (st < ed) {
            int v = min(height[st], height[ed]) * (ed - st);
            if (v > r) {
                r = v;
            }

            if (height[st] < height[ed]) {
                st += 1;
            }
            else {
                ed -= 1;
            }
        }

        return r;
    }
};
