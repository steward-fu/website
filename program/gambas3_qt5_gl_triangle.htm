程式語言 - GAMBAS - Qt 5.0 - OpenGL 1.0 - Draw Triangle


參考資訊：
https://gambaswiki.org/website/en/main.html

main/.src/FMain.form 

# Gambas Form File 3.0

{ Form Form
  MoveScaled(0,0,100,75)
  Text = ("main")
}

main/.src/FMain.class 

Private hGL As GLArea
Private Const W As Integer = 320
Private Const H as Integer = 240
 
Public Sub Form_Open()
    Me.Resize(W, H)
 
    hGL = New GLArea(Me) As "gl"
    hGL.Move(0, 0, W, H)
End
 
Public Sub gl_Open()
    gl.ClearColor(0, 0, 0, 1)
End
 
Public Sub gl_Resize()
    gl.Viewport(0, 0, W, H)
End
 
Public Sub gl_Draw()
    gl.Begin(gl.TRIANGLES)
        gl.Vertex2f(-0.5, -0.5)
        gl.Vertex2f( 0.5, -0.5)
        gl.Vertex2f( 0.0,  0.5)
    gl.End()
End

main/.project 

Startup=FMain
Component=gb.qt5
Component=gb.opengl
Component=gb.qt5.opengl

編譯、執行 

$ gbc3 -v main
$ gbs3 -c main
