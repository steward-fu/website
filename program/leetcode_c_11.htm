int min(int a, int b)
{
    return (a > b) ? b : a;
}

int maxArea(int *height, int heightSize)
{
    int r = 0;
    int v = 0;
    int st = 0;
    int ed = heightSize - 1;

    while (st < ed) {
        v = min(height[st], height[ed]) * (ed - st);
        if (v > r) {
            r = v;
        }

        if (height[st] > height[ed]) {
            ed -= 1;
        }
        else {
            st += 1;
        }
    }

    return r;
}
