main.c 

#include <stdio.h>
#include <jit/jit.h>

int main(int argc, char *argv[])
{
    const char *msg = "Hello, world!";

    jit_context_t context = jit_context_create();
    jit_context_build_start(context);
    jit_type_t signature = jit_type_create_signature(
        jit_abi_cdecl,
        jit_type_int,
        NULL,
        0,
        0
    );

    jit_function_t func = jit_function_create(context, signature);

    jit_type_t puts_params[1] = {
        jit_type_create_pointer(jit_type_sys_char, 1)
    };

    jit_type_t puts_sig = jit_type_create_signature(
        jit_abi_cdecl,
        jit_type_int,
        puts_params,
        1,
        0
    );

    jit_value_t arg = jit_value_create_nint_constant(
        func,
        jit_type_create_pointer(jit_type_sys_char, 1),
        (jit_nint)msg
    );
    jit_value_t args[1] = { arg };

    jit_insn_call_native(
        func,
        "puts",
        puts,
        puts_sig,
        args,
        1,
        0
    );

    jit_value_t zero = jit_value_create_nint_constant(func, jit_type_int, 0);
    jit_insn_return(func, zero);

    jit_function_compile(func);
    jit_context_build_end(context);

    int (*compiled_func)(void) = (int (*)(void))jit_function_to_closure(func);
    compiled_func();

    jit_context_destroy(context);
    return 0;
}
