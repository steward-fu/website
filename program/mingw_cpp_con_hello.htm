main.c 

#include <stdio.h>
#include <stdlib.h>
#include <windows.h>

int main(HINSTANCE hInstance, HINSTANCE hPrevInstance, LPSTR lpCmdLine, int nCmdShow)
{
    printf("Hello, world!\n");

    return 0;
}

編譯、執行 

$ i686-w64-mingw32-gcc -mconsole main.c -o main.exe
$ wine ./main.exe
    Hello, world!
