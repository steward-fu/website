解答： 

int largestAltitude(int *gain, int gainSize)
{
    int r = 0;
    int cc = 0;
    int sum = 0;

    for (cc = 0; cc < gainSize; cc++) {
        sum += gain[cc];
        if (sum > r) {
            r = sum;
        }
    }

    return r;
}
