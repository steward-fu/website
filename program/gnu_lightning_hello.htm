程式語言 - GNU - lightning - Hello, world!


參考資訊：
https://www.gnu.org/software/lightning/manual/lightning.html

main.c 

#include <stdio.h>
#include <lightning.h>

int main(int argc, char *argv[])
{
    void (*pfn)(void);
    jit_state_t *_jit = NULL;

    init_jit(argv[0]);
    _jit = jit_new_state();
 
    jit_prolog();
    jit_prepare();
    jit_pushargi((jit_word_t)"Hello, world!\n");
    jit_finishi(printf);
    jit_ret();
    jit_epilog();
 
    pfn = jit_emit();
    pfn();

    jit_clear_state(); 
    jit_destroy_state();
    finish_jit();

    return 0;
}
