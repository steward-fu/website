main.c 

#include <stdio.h>
#include <lightning.h>

static jit_state_t *_jit;

typedef void (*pvfi)(int);

int main(int argc, char *argv[])
{
    pvfi myFunction = NULL;
    jit_node_t *start = NULL;
    jit_node_t *end = NULL;

    init_jit(argv[0]);
    _jit = jit_new_state();

    start = jit_note(__FILE__, __LINE__);
    jit_prolog();
    jit_prepare();
    jit_pushargi((jit_word_t)"Hello, world!\n");
    jit_ellipsis();
    jit_pushargr(JIT_R1);
    jit_finishi(printf);
    jit_ret();
    jit_epilog();
    end = jit_note(__FILE__, __LINE__);

    myFunction = jit_emit();
    myFunction((char*)jit_address(end) - (char*)jit_address(start));
    jit_clear_state();

    jit_disassemble();

    jit_destroy_state();
    finish_jit();
    return 0;
}
