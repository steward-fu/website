main.c 

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

#include <lvgl.h>
#include <SDL2/SDL.h>

static lv_display_t *disp = NULL;

int main(int argc, char *argv[])
{
    lv_init();
    disp = lv_sdl_window_create(320, 240);

    lv_obj_t *label = lv_label_create(lv_screen_active());
    lv_label_set_text(label, "Hello, world!");

    lv_timer_handler();
    SDL_Delay(3000);

    return 0;
}

編譯、執行 

$ gcc main.c -o main -I/usr/local/include/lvgl/src -llvgl -lSDL2
$ ./main
