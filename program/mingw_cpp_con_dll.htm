main.c 

#include <stdio.h>

int test(const char *);

int main(int argc, char *argv[])
{
    test("test");
    return 0;
}

test.c 

#include <stdio.h>

int test(const char *buf)
{
    printf("%s\n", buf);
    return 0;
}

編譯、執行 

$ i686-w64-mingw32-gcc -mconsole -shared test.c -o test.dll
$ i686-w64-mingw32-gcc -mconsole main.c -o main.exe test.dll
$ wine ./main.exe
    test
