char* removeStars(char* s)
{
    int i = 0;
    int pos = 0;
    int len = strlen(s);

    for (i = 0; i < len; i++) {
        if (s[i] == '*') {
            if (pos > 0) {
                pos -= 1;
            }
        }
        else {
            s[pos++] = s[i];
        }
    }
    s[pos] = 0;

    return s;
}
