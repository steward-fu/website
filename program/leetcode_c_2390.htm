char* removeStars(char* s)
{
    int cc = 0;
    int idx = 0;
    int len = strlen(s);
    char *r = malloc(len + 1);

    memset(r, 0, len + 1);

    for (cc = 0; cc < len; cc++) {
        if (s[cc] != '*') {
            r[idx++] = s[cc];
        }
        else {
            if (idx > 0) {
                r[--idx] = 0;
            }
        }
    }

    return r;
}
