double findMaxAverage(int* nums, int numsSize, int k)
{
    int right = 0;
    double r = -INFINITY;
    double sum = 0.0;

    for (right = 0; right < numsSize; right++) {
        sum += nums[right];

        if (right >= (k - 1)) {
            if (sum > r) {
                r = sum;
            }

            sum -= nums[right - (k - 1)];
        }
    }

    return r / k;
}
