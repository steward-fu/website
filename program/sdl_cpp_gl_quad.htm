main.c 

#include <math.h>
#include <stdio.h>
#include <stdlib.h>
#include <SDL.h>
#include <GL/gl.h>

int main(int argc, char *argv[])
{
    const int W = 320;
    const int H = 240;
    GLdouble nearPlane = 0.1;
    GLdouble farPlane = 100.0;
    GLdouble top = nearPlane * tan(45.0 * 3.14159 / 360.0);
    GLdouble right = top * ((GLfloat)W / (GLfloat)H);
    SDL_Surface *screen = NULL;

    SDL_Init(SDL_INIT_VIDEO);
    screen = SDL_SetVideoMode(W, H, 16, SDL_SWSURFACE | SDL_OPENGL);

    glViewport(0, 0, W, H);

    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();
    glFrustum(-right, right, -top, top, nearPlane, farPlane);

    glMatrixMode(GL_MODELVIEW);

    glLoadIdentity();
    glPolygonMode(GL_FRONT_AND_BACK, GL_FILL);
    glTranslatef(-1.0f, 0.0f, -5.0f);
    glBegin(GL_QUADS);
        glColor3f(1.0f, 0.0f, 0.0f);
        glVertex2f(-0.5f, 0.5f);

        glColor3f(0.0f, 1.0f, 0.0f);
        glVertex2f(0.5f, 0.5f);

        glColor3f(0.0f, 0.0f, 1.0f);
        glVertex2f(0.5f, -0.5f);

        glColor3f(1.0f, 1.0f, 0.0f);
        glVertex2f(-0.5f, -0.5f);
    glEnd();

    glLoadIdentity();
    glPolygonMode(GL_FRONT_AND_BACK, GL_LINE);
    glTranslatef(1.0f, 0.0f, -5.0f);
    glBegin(GL_QUADS);
        glColor3f(1.0f, 0.0f, 0.0f);
        glVertex2f(-0.5f, 0.5f);

        glColor3f(0.0f, 1.0f, 0.0f);
        glVertex2f(0.5f, 0.5f);

        glColor3f(0.0f, 0.0f, 1.0f);
        glVertex2f(0.5f, -0.5f);

        glColor3f(1.0f, 1.0f, 0.0f);
        glVertex2f(-0.5f, -0.5f);
    glEnd();

    SDL_GL_SwapBuffers();
    SDL_Delay(3000);

    SDL_Quit();
    return 0;
}

編譯、執行 

$ gcc main.c -I/usr/include/SDL -o main -lSDL -lGL
$ ./main
