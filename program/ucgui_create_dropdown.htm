main.c 

#include "GUI.h"
#include "DROPDOWN.h"
  
int main(int argc, char *argv[])
{
    DROPDOWN_Handle hDrop = 0;

    GUI_Init();
    GUI_SetBkColor(GUI_GRAY);
    GUI_Clear();

    hDrop = DROPDOWN_Create(0, 100, 50, 100, 100, DROPDOWN_CF_AUTOSCROLLBAR | DROPDOWN_CF_UP);
    DROPDOWN_AddString(hDrop, "Test1");
    DROPDOWN_AddString(hDrop, "Test2");
    DROPDOWN_AddString(hDrop, "Test3");
    DROPDOWN_Expand(hDrop);
    GUI_Delay(1000);
    DROPDOWN_Delete(hDrop);

    GUI_Delay(3000);
    return 0;
}

編譯、執行 

$ gcc main.c -o main libucgui.a -IGUI_X -IGUI/Core -IGUI/Widget -IGUI/WM -lSDL
$ ./main
