程式語言 - uC/GUI - Create Button


參考資訊：
https://github.com/piyushpandey013/ucGUI

main.c 

#include "GUI.h"
#include "BUTTON.h"
 
int main(int argc, char *argv[])
{
    BUTTON_Handle hBtn = 0;
 
    GUI_Init();
    GUI_SetBkColor(GUI_GRAY);
    GUI_Clear();

    hBtn = BUTTON_Create(100, 100, 100, 30, 100, BUTTON_CF_SHOW);
    BUTTON_SetText(hBtn, "Test");
    BUTTON_SetFont(hBtn, &GUI_Font8x16);
    BUTTON_SetTextColor(hBtn, 0, GUI_BLUE);
    WM_SetFocus(hBtn);
    GUI_Delay(1000);
    BUTTON_Delete(hBtn);
 
    GUI_Delay(3000);
    return 0;
}

編譯、執行 

$ gcc main.c -o main libucgui.a -IGUI_X -IGUI/Core -IGUI/Widget -IGUI/WM -lSDL
$ ./main
