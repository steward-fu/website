class Solution {
public:
    int equalPairs(vector<vector<int>>& grid) {
        int i = 0;
        int j = 0;
        int r = 0;
        string col[200];
        string row[200];

        for (i = 0; i < grid.size(); i++) {
            for (j = 0; j < grid.size(); j++) {
                col[i] += std::to_string(grid[i][j]) + ",";
                row[i] += std::to_string(grid[j][i]) + ",";
            }
        }

        for (i = 0; i < grid.size(); i++) {
            for (j = 0; j < grid.size(); j++) {
                if (col[i] == row[j]) {
                    r += 1;
                }
            }
        }

        return r;
    }
};
