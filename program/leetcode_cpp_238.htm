class Solution {
public:
    vector<int> productExceptSelf(vector<int>& nums) {
        vector<int> r(nums.size());
        vector<int> f(nums.size(), 1);
        vector<int> b(nums.size(), 1);

        for (int i = 1; i < nums.size(); i++) {
            f[i] = f[i - 1] * nums[i - 1];
        }

        for (int i = nums.size() - 2; i >= 0; i--) {
            b[i] = b[i + 1] * nums[i + 1];
        }

        for (int i = 0; i < nums.size(); i++) {
            r[i] = f[i] * b[i];
        }

        return r;
    }
};
