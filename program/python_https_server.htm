server.py 

import sys
import ssl
import http.server

httpd = http.server.HTTPServer((sys.argv[1], int(sys.argv[2])), http.server.SimpleHTTPRequestHandler)
ctx = ssl.create_default_context(ssl.Purpose.CLIENT_AUTH)
ctx.load_cert_chain(certfile='server_cert.pem', keyfile='server_key.pem')
httpd.socket = ctx.wrap_socket(httpd.socket, server_side=True)
httpd.serve_forever()

client.py 

import sys
import ssl
import socket

s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
s.connect((sys.argv[1], int(sys.argv[2])))

s = ssl.wrap_socket(s, keyfile='client_key.pem', certfile='client_cert.pem', server_side=False)
s.sendall(f"GET / HTTP/1.1\r\nHost: github.com\r\nConnection: close\r\n\r\n".encode())

r = s.recv(1024)
if r != None:
    print(r)
s.close()
