#define MAX_SIZE 100000

typedef struct {
    int size;
    int buf[MAX_SIZE];
} queue;

void sort_left(queue *q, int index)
{
    int left = (index - 1) / 2;

    if (index && (q->buf[left] > q->buf[index])) {
        int t = q->buf[index];

        q->buf[index] = q->buf[left];
        q->buf[left] = t;

        sort_left(q, left);
    }
}

int sort_right(queue *q, int index)
{
    int s = index;
    int left = (2 * index) + 1;
    int right = (2 * index) + 2;

    if ((left < q->size) && (q->buf[left] < q->buf[s])) {
        s = left;
    }

    if ((right < q->size) && (q->buf[right] < q->buf[s])) {
        s = right;
    }

    if (s != index) {
        int t = q->buf[index];

        q->buf[index] = q->buf[s];
        q->buf[s] = t;

        sort_right(q, s);
    }

    return 0;
}

int sort_2d(const void *a, const void *b)
{
    return ((*(int **)b)[0] - (*(int **)a)[0]);
}

long long maxScore(int *nums1, int nums1Size, int *nums2, int nums2Size, int k)
{
    long long t = 0;
    long long r = INT_MIN;
    long long sum = 0;

    int cc = 0;
    int idx = 0;
    int **sorted = NULL;
    queue q = { 0 };

    sorted = malloc(sizeof(int *) * nums1Size);
    for (cc = 0; cc < nums1Size; cc++) {
        sorted[cc] = malloc(sizeof(int) * 2);
        sorted[cc][0] = nums2[cc];
        sorted[cc][1] = nums1[cc];
    }
    qsort(sorted, nums1Size, sizeof(int) * 2, sort_2d);

    for (cc = 0; cc < nums1Size; cc++) {
        sum += sorted[cc][1];
        q.buf[q.size++] = sorted[cc][1];
        sort_left(&q, q.size - 1);

        if (q.size > k) {
            sum -= q.buf[0];
            q.buf[0] = q.buf[--q.size];
            sort_right(&q, 0);
        }

        if (q.size == k) {
            t = sum * sorted[cc][0];
            if (t > r) {
                r = t;
            }
        }
    }

    return r;
}
