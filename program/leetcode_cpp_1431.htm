解答： 

class Solution {
public:
    vector<bool> kidsWithCandies(vector<int>& candies, int extraCandies) {
        vector<bool> ret;
        int max_num = *std::max_element(candies.begin(), candies.end());

        for (int i = 0; i < candies.size(); i++) {
            ret.push_back((candies[i] + extraCandies) >= max_num ? true : false);
        }

        return ret;
    }
};
