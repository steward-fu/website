class MyCircularDeque {
private:
    int sz;
    int rear;
    int front;
    int max_size;
    vector<int> data;

public:
    MyCircularDeque(int k) {
        sz = 0;
        rear = 0;
        front = 0;
        max_size = k;
        data.resize(k);
    }
    
    bool insertFront(int value) {
        if (isFull()) {
            return false;
        }

        front = (front - 1 + max_size) % max_size;
        data[front] = value;
        sz += 1;

        return true;
    }
    
    bool insertLast(int value) {
        if (isFull()) {
            return false;
        }

        data[rear] = value;
        rear = (rear + 1) % max_size;
        sz += 1;

        return true;
    }
    
    bool deleteFront() {
        if (isEmpty()) {
            return false;
        }

        front = (front + 1) % max_size;
        sz -= 1;
        return true;
    }
    
    bool deleteLast() {
        if (isEmpty()) {
            return false;
        }

        rear = (rear - 1 + max_size) % max_size;
        sz -= 1;
        return true;
    }
    
    int getFront() {
        if (isEmpty()) {
            return -1;
        }

        return data[front];
    }
    
    int getRear() {
        if (isEmpty()) {
            return -1;
        }

        return data[(rear - 1 + max_size) % max_size];
    }
    
    bool isEmpty() {
        return sz == 0;
    }
    
    bool isFull() {
        return sz == max_size;
    }
};
