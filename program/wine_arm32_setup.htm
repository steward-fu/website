程式語言 - Wine - Assembly (ARM 32) - 開發環境


參考資訊：
https://www.winehq.org/
https://wiki.winehq.org/Winelib_User%27s_Guide

安裝步驟(Debian 12 aarch64)： 

$ sudo apt-get install \
    gcc-mingw* \
    wine* \
    libwine-development:armhf=4.2-4+b1 \
    libwine-development-dev:armhf=4.2-4+b1 \
    wine32-development:armhf=4.2-4+b1

$ winegcc --version
    gcc (Debian 12.2.0-14+deb12u1) 12.2.0

$ cd
$ wget https://github.com/steward-fu/website/releases/download/wine/usr_lib_arm-linux-gnueabihf_wine-development.tar.gz
$ tar xvf usr_lib_arm-linux-gnueabihf_wine-development.tar.gz
$ sudo mv wine-development /usr/lib/arm-linux-gnueabihf
$ sudo ln -s /usr/lib/arm-linux-gnueabihf/wine-development/usr/lib/wine-development/winegcc /usr/local/bin/winegcc-armhf

$ sudo vim /usr/local/bin/mywinegcc-arm 
    #!/bin/bash
    winegcc-armhf -b arm-linux-gnueabihf "$@"

$ sudo vim /usr/local/bin/myarm32 
    #!/bin/bash
    WINEPREFIX=~/.wine_arm32 /usr/lib/wine-development/wine "$@"

$ sudo chmod a+x /usr/local/bin/myarm32
$ sudo chmod a+x /usr/local/bin/mywinegcc-arm

$ export WINEBUILD=/usr/lib/arm-linux-gnueabihf/wine-development/usr/lib/wine-development/winebuild
$ winegcc-armhf --version
    gcc (Debian 12.2.0-14+deb12u1) 12.2.0

$ cd
$ git clone https://github.com/steward-fu/wadk
$ sudo mv wadk /opt
