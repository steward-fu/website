bool isSubsequence(char* s, char* t)
{
    int cc = 0;
    int idx = 0;
    int s_len = strlen(s);
    int t_len = strlen(t);

    if (s_len == 0) {
        return true;
    }

    if (t_len == 0) {
        return false;
    }

    for (cc = 0; cc < t_len; cc++) {
        if (t[cc] == s[idx]) {
            idx += 1;
            if (idx >= s_len) {
                return true;
            }
        }
    }

    return false;
}
