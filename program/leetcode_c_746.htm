int min(int a, int b)
{
    return a > b ? b : a;
}

int minCostClimbingStairs(int *cost, int costSize)
{
    int cc = 0;
    int c0 = 0;
    int c1 = 0;

    for (cc = 0; cc < costSize; cc++) {
        int t = min(c0, c1) + cost[cc];

        c0 = c1;
        c1 = t;
    }

    return min(c0, c1);
}
