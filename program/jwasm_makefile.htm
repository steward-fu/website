程式語言 - JWasm - Makefile


TARGET   = main
MYWINE   = WINEPREFIX=~/.wine_x86 box86 wine
CFLAGS   = -coff
LDFLAGS  = -lgdi32
LDFLAGS += -luser32
LDFLAGS += -ladvapi32
LDFLAGS += -lkernel32
LDFLAGS += -e _start@0

all:
	jwasm ${CFLAGS} ${TARGET}.s
	i686-w64-mingw32-ld ${TARGET}.o -o ${TARGET}.exe ${LDFLAGS}

run:
	${MYWINE} ${TARGET}.exe

clean:
	rm -rf ${TARGET}.exe ${TARGET}.o
