一般在建立Windows視窗時，會註冊自定義的Class並且設定事件處理的副程式(callback)，不過，目前範例使用的Class是系統預設的WC_DIALOG，因此，可以透過SetWindowLong()設定callback，這種替換callback的行為就稱為Subclass，意指為原本Class的Subclass，所以Subclass後，需要將原先callback的位址(SetWindowLong()的回傳值)保存下來，然後透過呼叫CallWindowProc()(傳入原先callback的位址)，用來將處理控制權移交回原先視窗元件，所以當有多層Subclass的情形發生時，每次在處理事件時，記得判斷對應的Handle，如：在處理關閉事件時(WM_CLOSE)，會先判斷目前的hWnd是否為當前的視窗hWnd，如果是當前視窗hWnd，才進行關閉的動作，避免關閉到其它視窗元件

main.S 

    .include "/opt/wadk/wine/include/windows.inc"
  
    .data
hWin:           .skip 4
szMsg:          .skip 28
szAppName:      .asciz "main"
pDefWndProc:    .skip 4
 
    .balign 4
    .text
WndProc:
    push { lr }
 
    cmp r1, #WM_CLOSE
    beq 0f
    cmp r1, #WM_DESTROY
    beq 1f
    b 2f
   
0:
    bl DestroyWindow
    mov r0, #0
    b 3f
   
1:
    mov r0, #0
    bl PostQuitMessage
    mov r0, #0
    b 3f
   
2:
    push { r3 }
    mov r3, r2
    mov r2, r1
    mov r1, r0
    ldr r0, =pDefWndProc
    ldr r0, [r0]
    bl CallWindowProc
    add sp, sp, #4
    b 3f
   
3:
    pop { pc }
   
WinMain:
    push { lr }
   
    mov r3, #NULL
    mov r2, #NULL
    mov r1, #NULL
    mov r0, #NULL
    push { r0, r1, r2, r3 }
    mov r3, #240
    mov r2, #320
    mov r1, #0
    mov r0, #0
    push { r0, r1, r2, r3 }
    ldr r3, =WS_OVERLAPPEDWINDOW | WS_VISIBLE
    ldr r2, =szAppName
    mov r1, #WC_DIALOG
    mov r0, #WS_EX_LEFT
    bl CreateWindowEx
 
    add sp, sp, #32
    ldr r1, =hWin
    str r0, [r1]
   
    ldr r2, =WndProc
    ldr r1, =GWL_WNDPROC
    bl SetWindowLong
 
    ldr r1, =pDefWndProc
    str r0, [r1]
   
0:
    mov r3, #0
    mov r2, #0
    mov r1, #0
    ldr r0, =szMsg
    bl GetMessage
   
    cmp r0, #0
    beq 1f
     
    ldr r0, =szMsg
    bl DispatchMessage
    b 0b
   
1:
    mov r0, #0
    bl ExitProcess
 
    pop { pc }

編譯、執行 

$ mywinegcc-arm main.S -o main.exe
$ myarm32 main.exe.so
