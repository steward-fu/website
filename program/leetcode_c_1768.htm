char* mergeAlternately(char* word1, char* word2)
{
    int i = 0;
    int j = 0;
    int max_len = 0;
    int len1 = strlen(word1);
    int len2 = strlen(word2);
    char *ret = calloc(len1 + len2 + 1, sizeof(char));

    max_len = len1 > len2 ? len1 : len2;

    for (i = 0; i < max_len; i++) {
        if (i < len1) {
            ret[j++] = word1[i];
        }

        if (i < len2) {
            ret[j++] = word2[i];
        }
    }

    return ret;
}
