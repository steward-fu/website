int max(int a, int b)
{
    return a > b ? a : b;
}

char* mergeAlternately(char *word1, char *word2)
{
    int cc = 0;
    int idx = 0;
    int len = 0;
    int s1_len = strlen(word1);
    int s2_len = strlen(word2);
    char *r = malloc(s1_len + s2_len + 1);

    len = max(s1_len, s2_len);

    for (cc = 0; cc < len ;cc++) {
        if (cc < s1_len) {
            r[idx++] = word1[cc];
        }

        if (cc < s2_len) {
            r[idx++] = word2[cc];
        }
    }
    r[idx++] = 0;

    return r;
}
