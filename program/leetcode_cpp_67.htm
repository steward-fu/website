class Solution {
public:
    string addBinary(string a, string b) {
        string ans;
        int m = a.size() - 1;
        int n = b.size() - 1;
        int c = 0;

        while (m >= 0 || n >= 0) {
            int av = m >= 0 ? a[m--] - '0' : 0;
            int bv = n >= 0 ? b[n--] - '0' : 0;
            int sum = av + bv + c;

            ans = to_string(sum % 2) + ans;
            c = sum / 2;
        }

        return c == 1 ? "1" + ans : ans;
    }
};
