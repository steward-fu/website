#include <stdio.h>
#include <stdlib.h>
#include <stdarg.h>

static void mylog(const char *fmt, ...)
{
    FILE *file = fopen("/tmp/mylog.txt", "a+");

    if (file) {
        va_list va;

        va_start(va, fmt);
        vfprintf(file, fmt, va);
        va_end(va);

        fclose(file);
    }
}

int main(int argc, char **argv)
{
    mylog("DEBUG: %d, %s\n", 2024, "test");
    return 0;
}
