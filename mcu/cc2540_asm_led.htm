Alternative Function


main.s 

    p0sel set 0xf3
    p0dir set 0xfd
  
    .org 0h
    jmp _start
   
    .org 100h
_start:
    mov p0dir, #0x01
    mov p0sel, #0x00
loop:
    setb p0.0
    call delay
    clr p0.0
    call delay
    jmp loop

delay:
    mov r6, #50
d0:
    mov r5, #255
d1:
    mov r4, #255
d2:
    djnz r4, d2
    djnz r5, d1
    djnz r6, d0
    ret
    .end
