#include "stc15w4k56s4.h"

#define CS    P32
#define RST   P33
#define SCK   P34
#define SDA   P35
#define WR    P36

void delayms(unsigned int ms)
{
    unsigned int cnt = 0;

    while (ms--) {
        for (cnt = 0; cnt < 1000; cnt++) {
        }
    }
}

void gpio_init(void)
{
    P0M0 = 0x00;
    P0M1 = 0x00;
    P1M0 = 0x00;
    P1M1 = 0x00;
    P2M0 = 0x00;
    P2M1 = 0x00;
    P3M0 = 0x00;
    P3M1 = 0x00;
    P4M0 = 0x00;
    P4M1 = 0x00;

    P0 = 0xff;
    P1 = 0xff;
    P2 = 0xff;
    P3 = 0xff;
    P4 = 0xff;
}

void spi_cmd(unsigned char cmd)
{
    unsigned char bit = 0;

    CS = 0;
    SCK = 0;
    SDA = 0;
    SCK = 1;

    for (bit = 0; bit < 8; bit++) {
        SCK = 0;
        if ((cmd & 0x80) == 0x80) {
            SDA = 1;
        }
        else {
            SDA = 0;
        }
        SCK = 1;
        cmd <<= 1;
    }
    CS = 1;
}

void spi_dat(unsigned char dat)
{
    unsigned char bit = 0;

    CS = 0;
    SCK = 0;
    SDA = 1;
    SCK = 1;

    for (bit = 0; bit < 8; bit++) {
        SCK = 0;
        if ((dat & 0x80) == 0x80) {
            SDA = 1;
        }
        else {
            SDA = 0;
        }
        SCK = 1;
        dat <<= 1;
    }
    CS = 1;
}

void main(void)
{
    unsigned int x = 0;
    unsigend int y = 0;

    gpio_init();
    AUXR |= 0x80;

    WR = 0;
    RST = 0;
    delayms(300);
    RST = 1;
    delayms(300);

    lcd_init();

    spi_cmd(0x2c);
    for (y = 0; y < 240; y++) {
        for (x = 0; x < 240; x++) {
            spi_dat(0x7e);
            spi_dat(0x00);
        }
    }

    while (1) {
        P55 = 0;
        delayms(500);
        P55 = 1;
        delayms(500);
    }
}
