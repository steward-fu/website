main.c 

#include <stdio.h>
#include "pico/stdlib.h"

#define BTN 15
#define LED 25

void main(void)
{
    gpio_init(LED);
    gpio_init(BTN);
    gpio_set_dir(BTN, GPIO_IN);
    gpio_set_dir(LED, GPIO_OUT);
    gpio_pull_up(BTN);

    while (1) {
        gpio_put(LED, gpio_get(BTN));
    }
}
