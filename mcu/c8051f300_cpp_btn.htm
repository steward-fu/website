Button是連接到P0.6


Crossbar


XBR2


Watchdog


P0


main.c 

#include <mcs51/8051.h>

#define LED P0_0
#define BTN P0_6

__sfr __at(0xe3) XBR2;
__sfr __at(0xd9) PCA0MD;

void main(void)
{
    XBR2 = 0x40;
    PCA0MD = 0x00;

    while (1) {
        LED = BTN;
    }
}
