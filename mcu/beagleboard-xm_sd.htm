微處理器 - TI DM3730 (BeagleBoard-XM) - 如何製作可開機的MicroSD


$ sudo fdisk /dev/sdX
    Device     Boot Start   End Sectors Size Id Type
    /dev/sdb1  *     2048 26623   24576  12M  c W95 FAT32 (LBA)

$ sudo mkfs.vfat /dev/sdX1

$ wget https://github.com/steward-fu/website/releases/download/beagleboard-xm/MLO
$ wget https://github.com/steward-fu/website/releases/download/beagleboard-xm/u-boot.img
$ sudo mount /dev/sdX1 /mnt
$ cp MLO u-boot.img /mnt
$ sudo umount /mnt
