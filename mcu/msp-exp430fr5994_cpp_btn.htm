main.c 

#include <msp430fr5994.h>
 
int main(void)
{
    WDTCTL = WDTPW + WDTHOLD;
    PM5CTL0 &= ~LOCKLPM5;
    P1DIR = BIT0;
    P5REN = BIT5;
    P5OUT = BIT5;
 
    while (1) {
        P1OUT = ~(P5IN >> 5);
    }
}
