微處理器 - Allwinner V3S (Lichee Pi Zero) - Build Kernel 6.3.5


$ cd
$ wget https://cdn.kernel.org/pub/linux/kernel/v6.x/linux-6.3.5.tar.xz
$ tar xvf linux-6.3.5.tar.xz
$ cd linux-6.3.5
$ make ARCH=arm CROSS_COMPILE=arm-linux-gnueabihf- sunxi_defconfig
$ make ARCH=arm CROSS_COMPILE=arm-linux-gnueabihf- zImage modules dtbs -j4
$ make ARCH=arm CROSS_COMPILE=arm-linux-gnueabihf- INSTALL_MOD_PATH=out modules_install -j4
