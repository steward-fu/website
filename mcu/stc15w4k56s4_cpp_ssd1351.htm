void delayms(unsigned int ms)
{
    unsigned int cnt = 0;

    while (ms--) {
        for (cnt = 0; cnt < 1000; cnt++) {
        }
    }
}

void CMD(char cmd)
{
    int i = 0;

    CS = 0;
    DC = 0;
    for (i = 0; i < 8; i++) {
        SCK = 0;
        if ((cmd & 0x80) == 0x80) {
            SDI = 1;
        }
        else {
            SDI = 0;
        }
        cmd <<= 1;
        SCK = 1;
    }
    CS = 1;
}

void DATA(char dat)
{
    int i = 0;

    CS = 0;
    DC = 1;
    for (i = 0; i < 8; i++) {
        SCK = 0;
        if ((dat & 0x80) == 0x80) {
            SDI = 1;
        }
        else {
            SDI = 0;
        }
        dat <<= 1;
        SCK = 1;
    }
    CS = 1;
}

void cursor(unsigned char x, unsigned char y)
{
    CMD(SSD1351_CMD_WRITERAM);
    CMD(SSD1351_CMD_SETCOLUMNADDRESS);
    DATA(x);
    DATA(127);

    CMD(SSD1351_CMD_SETROWADDRESS);
    DATA(y);
    DATA(127);
    CMD(SSD1351_CMD_WRITERAM);
}

void reset(void)
{
    RST = 0;
    delayms(100);
    RST = 1;
    delayms(100);
}

void color(void)
{
    int i, j;
    unsigned char r = 0x00;
    unsigned char g = 0x00;
    unsigned char b = 0xff;
    unsigned char data1 = (r & 0xF8) | (g >> 5);
    unsigned char data2 = (b >> 3) | ((g >> 2) << 5);

    cursor(0, 0);
    for (i = 0; i < 128; i++) {
        for (j = 0; j < 128; j++) {
            DATA(data1);
            DATA(data2);
        }
    }
}

void gpio_init(void)
{
    P0M0 = 0x00;
    P0M1 = 0x00;
    P1M0 = 0x00;
    P1M1 = 0x00;
    P2M0 = 0x00;
    P2M1 = 0x00;
    P3M0 = 0x00;
    P3M1 = 0x00;
    P4M0 = 0x00;
    P4M1 = 0x00;
    P5M0 = 0x00;
    P5M1 = 0x00;
}

void main(void)
{
    gpio_init();
    AUXR |= 0x80;

    reset();
    init();
    color();
    while (1) {
        P55 = 0;
        delayms(1000);
        P55 = 1;
        delayms(1000);
    }
}
