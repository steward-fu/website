main.c 

#include "stc15w4k56s4.h"

#define RST   P54
#define CS    P43
#define SCL   P42
#define WR    P41
#define SDA   P40

#define Command   0
#define Parameter 1

void delayms(unsigned int ms)
{
    unsigned int cnt = 0;

    while (ms--) {
        for (cnt = 0; cnt < 1000; cnt++) {
        }
    }
}

void Write(unsigned char flag, unsigned char dat)
{
    int i = 0;

    CS = 0;
    SCL = 0;
    SDA = flag;
    SCL = 1;

    for (i = 0; i < 8; i++) {
        SCL = 0;
        if ((dat & 0x80) == 0x80) {
            SDA = 1;
        }
        else {
            SDA = 0;
        }
        dat <<= 1;
        SCL = 1;
    }
    CS = 1;
}

void reset(void)
{
  RST = 0;
  delayms(100);
  RST = 1;
  delayms(100);
}

void color(void)
{
    unsigned int x = 0;
    unsigned int y = 0;

    Write(Command, 0x2c); 
    for (y = 0; y < 240; y++) {
        for (x = 0; x < 240; x++) {
            Write(Parameter, 0xf1);
            Write(Parameter, 0x00);
        }
    }
}

void gpio_init(void)
{
    P0M0 = 0x00;
    P0M1 = 0x00;
    P1M0 = 0x00;
    P1M1 = 0x00;
    P2M0 = 0x00;
    P2M1 = 0x00;
    P3M0 = 0x00;
    P3M1 = 0x00;
    P4M0 = 0x00;
    P4M1 = 0x00;
    P5M0 = 0x00;
    P5M1 = 0x00;
}

void main(void)
{
    gpio_init();
    AUXR |= 0x80;

    reset();
    init();
    color();

    while (1) {
        P55 = 0;
        delayms(1000);
        P55 = 1;
        delayms(1000);
    }
}
