main.s 

    p0sel set 0xf3
    p0dir set 0xfd
  
    .org 0h
    jmp _start
   
    .org 100h
_start:
    mov p0dir, #0x01
    mov p0sel, #0x00
loop:
    mov c, p0.1
    mov p0.0, c
    jmp loop
    .end
