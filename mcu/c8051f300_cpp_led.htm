LED是連接到P0.0


Crossbar


XBR2


Watchdog


P0


main.c 

#include <stdio.h>

__sfr __at(0x80) P0;
__sfr __at(0xe3) XBR2;
__sfr __at(0xd9) PCA0MD;

void delay(unsigned long cnt)
{
    while (cnt--) {
    }
}

void main(void)
{
    XBR2 = 0x40;
    PCA0MD = 0x00;

    while (1) {
        P0 ^= 1;
        delay(100000);
    }
}
