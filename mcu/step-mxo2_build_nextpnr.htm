$ sudo apt-get install libeigen3-dev libboost-all-dev

$ cd
$ git clone https://github.com/YosysHQ/nextpnr
$ cd nextpnr
$ git submodule update --init --recursive
$ mkdir build
$ cd build
$ cmake -DARCH=machxo2 -DTRELLIS_INSTALL_PREFIX=/usr/local ..
$ make -j4
$ sudo make install
