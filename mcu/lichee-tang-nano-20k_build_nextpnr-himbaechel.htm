$ pip install apycula

$ cd
$ git clone https://github.com/YosysHQ/nextpnr --recursive
$ mkdir nextpnr/build
$ cd nextpnr/build
$ cmake .. -DARCH="himbaechel" -DHIMBAECHEL_GOWIN_DEVICES="all"
$ make -j4
$ sudo make install
