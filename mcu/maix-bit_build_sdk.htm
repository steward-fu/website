微處理器 - Kendryte K210 (Sipeed Maix Bit) - Build SDK


$ cd
$ wget https://s3.cn-north-1.amazonaws.com.cn/dl.kendryte.com/documents/kendryte-toolchain-ubuntu-amd64-8.2.0-20190213.tar.gz
$ tar xvf kendryte-toolchain-ubuntu-amd64-8.2.0-20190213.tar.gz
$ sudo mv kendryte-toolchain /opt/k210
$ export LD_LIBRARY_PATH=/opt/k210/bin

$ wget https://s3.cn-north-1.amazonaws.com.cn/dl.kendryte.com/documents/kendryte-standalone-sdk-0.5.6.zip
$ unzip kendryte-standalone-sdk-0.5.6.zip
$ cd kendryte-standalone-sdk-0.5.6
$ mkdir build
$ cd build
$ cmake .. -DPROJ=hello_world -DTOOLCHAIN=/opt/k210/bin && make
