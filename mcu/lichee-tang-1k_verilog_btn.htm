main.v 

module main (
    input btn,
    output reg led
);
 
initial begin
    led = 0;
end
   
always @(posedge btn) begin
    led <= ~led;
end
  
endmodule

main.cst 

IO_LOC "led" 9;
IO_LOC "btn" 44;
