微處理器 - Microchip PIC10F200 - Assembly - Button


由於司徒沒有焊接提昇電阻，因此，需要設定OPTION暫存器


main.s 

    list p=10f200, r=hex
    #include <p10f200.inc>
    __config _CONFIG, _IntRC_OSC & _WDTE_OFF & _MCLRE_OFF
 
    org 0x00
start:
    movlw b'00000000'
    option
  
    movlw b'00001000'
    tris GPIO
    bsf GPIO, 0
    bsf GPIO, 1
    bsf GPIO, 2
    movlw b'00001100'
    tris GPIO
 
loop:
    btfss GPIO, 3
    bcf GPIO, 1
    btfsc GPIO, 3
    bsf GPIO, 1
    goto loop
    end

編譯 

$ gpasm main.s
