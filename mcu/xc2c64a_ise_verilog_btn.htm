main.v 

module main (
    input btn,
    output reg led
);
   
initial begin
    led = 0;
end
     
always @(*) begin
    led <= ~btn;
end

endmodule

main.ucf 

NET "btn" LOC = "P18";
NET "led" LOC = "P39";
