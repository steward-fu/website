main.vhd 

library ieee;
use ieee.std_logic_1164.all;
use ieee.numeric_std.all;
   
entity main is
    port(
        btn : in bit;
        led : out unsigned(7 downto 0)
    );
end main;
   
architecture logic of main is
    signal cnt : unsigned(7 downto 0) := "11111110";
  
begin
    process(btn) is
    begin
        if (btn'event and btn = '0') then
            cnt <= cnt rol 1;
        end if;
    end process;
    
    led <= cnt;
end logic;
