main.v 

module main (
    input clk,
    output reg led
);
 
reg [23:0] cnt;
     
always @(posedge clk) begin
    if (cnt == 5000000) begin
        cnt = 0;
        led <= ~led;
    end else
        cnt = cnt + 1;
end

endmodule

NET "clk"  LOC = "P1" | BUFG = CLK; 
NET "led"  LOC = "P39";
