家用機 - Sega Genesis - Build SGDK


參考資訊：
https://github.com/Stephane-D/SGDK
https://github.com/iratahack/m68k-elf-gcc/releases

$ cd
$ wget https://github.com/iratahack/m68k-elf-gcc/releases/download/v13.2.0_latest/m68k-elf.tgz
$ tar xvf m68k-elf.tgz
$ sudo mv m68k-elf /opt
$ export PATH=/opt/m68k-elf/bin:$PATH
$ m68k-elf-gcc --version
    m68k-elf-gcc (crosstool-NG UNKNOWN) 13.2.0

$ cd
$ wget https://download.oracle.com/java/24/latest/jdk-24_linux-x64_bin.tar.gz
$ tar xvf jdk-24_linux-x64_bin.tar.gz -C ~/
$ export JAVA_HOME=~/jdk-24.0.1
$ export CLASSPATH=.:$JAVA_HOME/lib/tools.jar:$JAVA_HOME/lib/dt.jar
$ export PATH=$JAVA_HOME/bin:$PATH

$ sudo chmod 0777 /opt
$ cd /opt
$ git clone https://github.com/Stephane-D/SGDK sgdk
$ mkdir /opt/sgdk/build
$ cd /opt/sgdk/build
$ cmake ..
$ sed -i 's/\/usr\/bin\/gcc/\/opt\/m68k-elf\/bin\/m68k-elf-gcc/g' CMakeFiles/md.dir/build.make
$ make

$ cd /opt/sgdk
$ make -f makefile.gen
$ make -f makelib.gen
