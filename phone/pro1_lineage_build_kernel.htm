$ cd
$ wget https://android.googlesource.com/platform/prebuilts/clang/host/linux-x86/+archive/refs/heads/main/clang-r536225.tar.gz

$ git clone https://github.com/LineageOS/android_kernel_fxtec_sm6115 -b lineage-22.2 --depth=1
$ cd android_kernel_fxtec_sm6115
$ mkdir clang
$ cd clang
$ tar xvf ~/clang-r536225.tar.gz
$ cd ..

$ sudo fallocate -l 16G /swapfile
$ sudo chmod 600 /swapfile
$ sudo mkswap /swapfile
$ sudo swapon /swapfile

$ export PATH=$PWD/clang/bin:$PATH
$ export ARCH=arm64
$ export CROSS_COMPILE=aarch64-linux-gnu-
$ export CROSS_COMPILE_COMPAT=arm-linux-gnueabi-
$ export LLVM=1
$ export LLVM_IAS=1
