module outer_shape() {
    hull() {
        translate([5, -2, 0]) cylinder(h, 5, 5);
        translate([86, -2, 0]) cylinder(h, 5, 5);
        translate([5, 139, 0]) cylinder(h, 5, 5);
        translate([86, 139, 0]) cylinder(h, 5, 5);
    }
}
